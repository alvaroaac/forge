# Phase 1 MVP Implementation Plan

> **Status:** Approved (2026-05-13). Plan is cleared for execution under `subagent-driven-development`. The plan addendum (if added later) lives at `thoughts/tasks/phase1-mvp/plans/2026-05-12-phase1-mvp.addendum.md`.

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Open app → see assigned Linear issues grouped by status/category → click an issue → stream a generated spec in the drawer → save to `thoughts/tasks/[id]/initial-spec.md`. Target: under 30s from launch to spec read.

**Architecture:** Electron app, strict main/renderer separation. Main process owns Linear polling, Claude API calls, filesystem I/O, auth checks. Renderer is React + TS, talks only through typed `window.forge` API exposed via `contextBridge`. Storage is flat JSON in `~/.forge/`. UI matches the Claude Design HTML/CSS prototype under `resources/design/forge/`.

**Tech Stack:** Electron 31, Vite 5 (renderer bundler), `electron-vite` for build orchestration, React 18 + TypeScript 5, Vitest for unit tests, Playwright (`@playwright/test`) + `electron` driver for one e2e smoke test, `@anthropic-ai/sdk` for Claude API streaming, existing `createLinearClient` from `.agents/skills/linear/reference/linear.mjs`.

## Design References (read before every UI task)

The Claude Design HTML/CSS/JS prototype is the source of truth for visual structure. Every component-port task names the prototype file + line range it mirrors. Read the cited region of the prototype, then apply the **Deltas from prototype** list before implementing. Do not invent class names, dimensions, or DOM structure — they are already in the prototype and must match.

| Prototype file | What it defines | Used by |
|---|---|---|
| `resources/design/forge/project/forge/styles.css` | Design tokens, every CSS class. Ported verbatim to `src/renderer/styles/tokens.css`. | Task 30 |
| `resources/design/forge/project/forge/icons.jsx` | Inline-SVG icon set, 20 named icons. | Task 31 |
| `resources/design/forge/project/forge/dashboard.jsx` | Top-level layout + every left/right-panel component (`StatusDot`, `PillTab`, `PriorityChip`, `LabelBadge`, `IssueCard`, `IssueGroup`, `IssueListPanel`, `AuthRow`, `ActivityRow`, `AgentCard`, `RightPanel`, `TopBar`, `Dashboard`). | Tasks 32–37, 45 |
| `resources/design/forge/project/forge/drawer.jsx` | Spec drawer overlay (`MarkdownSection`, `highlightInline`, `DetailTab`, `SpecTab`, `SpecDrawer`). | Tasks 38–41 |
| `resources/design/forge/project/forge/data.js` | Sample data shape — informs how the prototype expects `Issue`/`Spec` to look. **Do not port** — the real shapes live in `src/shared/types`. | Reference only |

**General delta pattern across all UI tasks:**
- Prototype uses denormalized fields (`issue.group`, `issue.label`, `issue.specStatus`) and a `window.FORGE_DATA` mock. The real `Issue` shape (`src/shared/types`) has `labels: string[]`, `isBug`, and no `group`/`specStatus`. UI derives group via `classifyGroup` and spec presence via a `hasSpecFor` prop fed from the renderer state.
- Prototype attaches globals via `Object.assign(window, {...})`. None of that is ported — every component is a named export.
- All untyped destructure props become explicit TypeScript prop interfaces.

## Conventions Recap (apply to every task)

- TDD: failing test before implementation. Each task has a "verify it fails" step.
- Max cyclomatic complexity 4 per function. Use early returns.
- No `any` without `// reason: ...` on the same line.
- Extract on the 2nd duplicate, not the 3rd.
- Commit after every passing task. Conventional Commits (`feat:`, `test:`, `chore:`, `refactor:`).
- Renderer never imports `node:*`. No `ipcRenderer` exposed — only `window.forge`.

```ts
import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { existsSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { homedir } from 'node:os';
import type { AppConfig } from '../../shared/types';

const DEFAULTS: AppConfig = {
  linearTokenPath: join(homedir(), '.humanlayer', 'riptide', 'linear.json'),
  linearTeamKey: 'FUL',
  repoPath: '',
  claudeModel: 'claude-sonnet-4-6',
};

export interface ConfigStore {
  get(): Promise<AppConfig>;
  set(patch: Partial<AppConfig>): Promise<void>;
}

async function readMerged(path: string): Promise<AppConfig> {
  if (!existsSync(path)) return { ...DEFAULTS };
  const raw = await readFile(path, 'utf-8');
  const parsed = JSON.parse(raw) as Partial<AppConfig>;
  return { ...DEFAULTS, ...parsed };
}

export function createConfigStore(path: string): ConfigStore {
  return {
    async get() { return readMerged(path); },
    async set(patch) {
      const current = await readMerged(path);
      const merged = { ...current, ...patch };
      await mkdir(dirname(path), { recursive: true });
      await writeFile(path, JSON.stringify(merged, null, 2), 'utf-8');
    },
  };
}
```

```ts
import { existsSync } from 'node:fs';
import { dirname } from 'node:path';

export interface IssuesCache {
  read(): Promise<Issue[]>;
  write(issues: Issue[]): Promise<void>;
}

export function createIssuesCache(path: string): IssuesCache {
  return {
    async read() {
      if (!existsSync(path)) return [];
      const raw = await readFile(path, 'utf-8');
      return JSON.parse(raw) as Issue[];
    },
    async write(issues) {
      await mkdir(dirname(path), { recursive: true });
      await writeFile(path, JSON.stringify(issues, null, 2), 'utf-8');
    },
  };
}
```

```ts
import { readFile, readdir, stat } from 'node:fs/promises';
import { existsSync } from 'node:fs';
import { join } from 'node:path';

export interface RepoContext {
  agentsMd: string;
  thoughts: Array<{ name: string; content: string }>;
}

async function readIfExists(path: string): Promise<string> {
  if (!existsSync(path)) return '';
  return readFile(path, 'utf-8');
}

async function readAgentsContext(repoPath: string): Promise<string> {
  const agents = await readIfExists(join(repoPath, 'AGENTS.md'));
  if (agents) return agents;
  return readIfExists(join(repoPath, 'CLAUDE.md'));
}

async function listThoughtFiles(thoughtsDir: string): Promise<string[]> {
  if (!existsSync(thoughtsDir)) return [];
  const entries = await readdir(thoughtsDir);
  const result: string[] = [];
  for (const name of entries) {
    if (!name.endsWith('.md')) continue;
    const full = join(thoughtsDir, name);
    const s = await stat(full);
    if (s.isFile()) result.push(name);
  }
  return result;
}

export async function readRepoContext(repoPath: string): Promise<RepoContext> {
  const agentsMd = await readAgentsContext(repoPath);
  const thoughtsDir = join(repoPath, 'thoughts');
  const names = await listThoughtFiles(thoughtsDir);
  const thoughts = await Promise.all(
    names.map(async (name) => ({
      name,
      content: await readFile(join(thoughtsDir, name), 'utf-8'),
    })),
  );
  return { agentsMd, thoughts };
}
```

```ts
export interface StreamSpecInput {
  client: Anthropic;
  model: string;
  system: string;
  user: string;
  onChunk: (delta: string) => void;
}

function extractDelta(event: { type: string; delta?: { type: string; text?: string } }): string {
  if (event.type !== 'content_block_delta') return '';
  if (event.delta?.type !== 'text_delta') return '';
  return event.delta.text ?? '';
}

export async function streamSpec(input: StreamSpecInput): Promise<string> {
  const stream = input.client.messages.stream({
    model: input.model,
    max_tokens: 2048,
    system: input.system,
    messages: [{ role: 'user', content: input.user }],
  });
  let full = '';
  for await (const event of stream) {
    const delta = extractDelta(event as never);
    if (!delta) continue;
    full += delta;
    input.onChunk(delta);
  }
  return full;
}
```

```ts
import type { Issue, SpecStreamChunk } from '../../shared/types';
import type { IssuesCache } from '../services/issues-cache';
import type { RepoContext } from '../services/repo-reader';
import Anthropic from '@anthropic-ai/sdk';

export interface SpecGenerateDeps {
  store: ConfigStore;
  cache: IssuesCache;
  readRepoContext: (repoPath: string) => Promise<RepoContext>;
  streamSpec: (opts: {
    client: Anthropic;
    model: string;
    system: string;
    user: string;
    onChunk: (delta: string) => void;
  }) => Promise<string>;
  writeSpec: (opts: { repoPath: string; issueId: string; content: string }) => Promise<string>;
  anthropic: Anthropic;
  templateMd: string;
}

function findIssue(issues: Issue[], id: string): Issue {
  const found = issues.find((i) => i.id === id);
  if (!found) throw new Error(`Issue not found in cache: ${id}`);
  return found;
}

export function registerSpecGenerateHandler(ipc: IpcMain, deps: SpecGenerateDeps): void {
  ipc.handle(IpcChannel.SpecGenerate, async (event, payload: { issueId: string }) => {
    const cfg = await deps.store.get();
    const issues = await deps.cache.read();
    const issue = findIssue(issues, payload.issueId);
    const context = await deps.readRepoContext(cfg.repoPath);
    const prompt = buildSpecPrompt({ issue, context, templateMd: deps.templateMd });
    const send = (chunk: SpecStreamChunk) => event.sender.send(IpcChannel.SpecStreamChunk, chunk);
    const content = await deps.streamSpec({
      client: deps.anthropic,
      model: cfg.claudeModel,
      system: prompt.system,
      user: prompt.user,
      onChunk: (delta) => send({ issueId: payload.issueId, delta, done: false }),
    });
    send({ issueId: payload.issueId, delta: '', done: true });
    await deps.writeSpec({ repoPath: cfg.repoPath, issueId: payload.issueId, content });
    return { content, issueId: payload.issueId };
  });
}
```

```tsx
import type { Issue } from '../../shared/types';
import { IssueCard } from './issue-card';
import { IconBug, IconFlame, IconSpark, IconTerminal } from './icons';
import type { Group } from '../lib/classify';

const META: Record<Group, { color: string; Icon: React.FC<{ size?: number; stroke?: number; style?: React.CSSProperties }> }> = {
  Bugs:    { color: 'var(--danger)', Icon: IconBug },
  Urgent:  { color: 'var(--warn)',   Icon: IconFlame },
  Feature: { color: 'var(--accent)', Icon: IconSpark },
  Chore:   { color: 'var(--text-3)', Icon: IconTerminal },
};

function intoRows<T>(items: T[]): [T[], T[]] {
  const rows: [T[], T[]] = [[], []];
  items.forEach((it, i) => rows[i % 2].push(it));
  return rows;
}

export function IssueGroup(props: {
  name: Group;
  items: Issue[];
  onOpen: (issue: Issue, which: 'spec' | 'detail') => void;
  activeId: string | null;
  hasSpecFor: (issueId: string) => boolean;
}) {
  const { name, items, onOpen, activeId, hasSpecFor } = props;
  const meta = META[name];
  const Ico = meta.Icon;
  const rows = intoRows(items);
  return (
    <div className="group">
      <div className="group-head">
        <span className="group-head-left">
          <Ico size={11} stroke={2} style={{ color: meta.color }} />
          <span className="group-name">{name}</span>
          <span className="group-count mono">{items.length}</span>
        </span>
      </div>
      <div className="group-scroller-wrap">
        <div className="group-scroller">
          {rows.map((row, ri) => (
            <div className="group-row" key={ri}>
              {row.map((iss) => (
                <IssueCard
                  key={iss.id}
                  issue={iss}
                  onOpen={onOpen}
                  isActive={iss.id === activeId}
                  hasSpec={hasSpecFor(iss.id)}
                />
              ))}
            </div>
          ))}
```

```tsx
import { classifyGroup, type Group } from '../lib/classify';

const TAB_KEY: Record<Tab, IssueStatus> = {

const GROUP_ORDER: Group[] = ['Bugs', 'Urgent', 'Feature', 'Chore'];

function counts(issues: Issue[]): Record<Tab, number> {
  return TABS.reduce((acc, t) => {
    acc[t] = issues.filter((i) => i.status === TAB_KEY[t]).length;
    return acc;
  }, {} as Record<Tab, number>);
}

function groupVisible(visible: Issue[]): Array<{ name: Group; items: Issue[] }> {
  return GROUP_ORDER
    .map((g) => ({ name: g, items: visible.filter((i) => classifyGroup(i) === g) }))
    .filter((g) => g.items.length > 0);
}

export function IssueListPanel(props: {
```

```ts
export function highlightInline(s: string): string {
  return s
    .replace(/`([^`]+)`/g, '<code class="md-code">$1</code>')
    .replace(/(§\d+(?:\.\d+)?)/g, '<span class="md-ref">$1</span>')
    .replace(/@(\w+)/g, '<span class="md-mention">@$1</span>');
}

export interface Section { h: string; body: string; }

export function splitSections(md: string): Section[] {
  const lines = md.split('\n');
  const sections: Section[] = [];
  let current: Section = { h: '', body: '' };
  for (const line of lines) {
    const m = /^##\s+(.+)$/.exec(line);
    if (m) {
      if (current.h || current.body) sections.push(current);
      current = { h: m[1], body: '' };
    } else {
      current.body += (current.body ? '\n' : '') + line;
    }
  }
  if (current.h || current.body.trim()) sections.push(current);
  return sections.length ? sections.map(trimBody) : [{ h: '', body: md }];
}

function trimBody(s: Section): Section { return { h: s.h, body: s.body.trim() }; }
```

```tsx
import { highlightInline } from '../lib/markdown';

function renderLine(line: string, key: number) {
  const trimmed = line.trim();
  if (!trimmed) return null;
  const numMatch = /^(\d+)\.\s+(.*)$/.exec(trimmed);
  const bulletMatch = /^[•\-*]\s+(.*)$/.exec(trimmed);
  if (numMatch) return <li className="md-li" key={key}><span className="md-li-mark mono">{numMatch[1]}.</span><span dangerouslySetInnerHTML={{ __html: highlightInline(numMatch[2]) }} /></li>;
  if (bulletMatch) return <li className="md-li" key={key}><span className="md-li-mark mono">•</span><span dangerouslySetInnerHTML={{ __html: highlightInline(bulletMatch[1]) }} /></li>;
  return <p key={key} dangerouslySetInnerHTML={{ __html: highlightInline(trimmed) }} />;
}

export function MarkdownSection({ h, body }: { h: string; body: string }) {
  const lines = body.split('\n');
  return (
    <section className="md-section">
      {h && <h3 className="md-h">{h}</h3>}
      <div className="md-body">{lines.map(renderLine)}</div>
    </section>
  );
}
```

```tsx
function useEscClose(onClose: () => void): void {
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') onClose(); };
```

```ts
import { useCallback, useEffect, useState } from 'react';
import type { Issue } from '../../shared/types';

const POLL_MS = 60_000;

export function useIssues() {
  const [issues, setIssues] = useState<Issue[]>([]);
  const [lastSync, setLastSync] = useState<number>(0);

  const refresh = useCallback(async () => {
    const next = await window.forge.linear.refresh();
    setIssues(next);
    setLastSync(Date.now());
  }, []);

  useEffect(() => {
    let cancelled = false;
    void window.forge.linear.fetch().then((cached) => {
      if (!cancelled) setIssues(cached);
      void refresh();
    });
    const id = setInterval(() => { void refresh(); }, POLL_MS);
    return () => { cancelled = true; clearInterval(id); };
  }, [refresh]);

  return { issues, lastSync, refresh };
}
```

```ts
import { useCallback, useEffect, useState } from 'react';
import type { Spec, SpecStreamChunk } from '../../shared/types';

export function useSpecStream(issueId: string | null) {
  const [spec, setSpec] = useState<Spec | null>(null);
  const [streaming, setStreaming] = useState('');
  const [isStreaming, setIsStreaming] = useState(false);

  useEffect(() => {
    if (!issueId) { setSpec(null); setStreaming(''); return; }
    void window.forge.spec.get(issueId).then(setSpec);
    setStreaming('');
    setIsStreaming(false);
  }, [issueId]);

  useEffect(() => {
    if (!issueId) return;
    return window.forge.spec.onChunk((chunk: SpecStreamChunk) => {
      if (chunk.issueId !== issueId) return;
      if (chunk.done) { setIsStreaming(false); return; }
      setStreaming((s) => s + chunk.delta);
    });
  }, [issueId]);

  const generate = useCallback(async () => {
    if (!issueId) return;
    setStreaming('');
    setIsStreaming(true);
    const result = await window.forge.spec.generate(issueId);
    setSpec({ issueId, content: result.content, generatedAt: new Date().toISOString(), approved: false });
  }, [issueId]);

  return { spec, streaming, isStreaming, generate };
}
```
